14def reorder_point(product: Product, weekly_sales: list[int] | None = None) -> int:
15 if product.lead_days <= 0:
16 raise ValueError("lead_days must be positive")
17
18 if weekly_sales:
19 daily = mean(weekly_sales[-WINDOW_WEEKS:]) / 7
